Table 1: BillTbl

CREATE TABLE [dbo].[BillTbl] (
    [BillId]   INT           IDENTITY (2100, 1) NOT NULL,
    [BillDate] DATETIME      NOT NULL,
    [Seller]   INT           NOT NULL,
    [SelName]  VARCHAR (MAX) NOT NULL,
    [Amount]   INT           NOT NULL,
    CONSTRAINT [PK_BillTbl] PRIMARY KEY CLUSTERED ([BillId] ASC),
    CONSTRAINT [FK_Seller] FOREIGN KEY ([Seller]) REFERENCES [dbo].[SellerTbl] ([SelId])
);




Table 2: CategoryTbl

CREATE TABLE [dbo].[CategoryTbl] (
    [CatId]          INT           IDENTITY (100, 1) NOT NULL,
    [CatName]        VARCHAR (50)  NOT NULL,
    [CatDescription] VARCHAR (MAX) NOT NULL,
    CONSTRAINT [PK_CategoryTbl] PRIMARY KEY CLUSTERED ([CatId] ASC)
);




Table 3: ProductSells

CREATE TABLE [dbo].[ProductSells] (
    [Id]        INT           IDENTITY (1001, 1) NOT NULL,
    [Sel_name]  VARCHAR (MAX) NOT NULL,
    [Bill_no]   INT           NOT NULL,
    [Product]   VARCHAR (MAX) NOT NULL,
    [Price]     INT           NOT NULL,
    [Quantity]  INT           NOT NULL,
    [Total]     INT           NOT NULL,
    [Date/Time] DATETIME      NOT NULL,
    PRIMARY KEY CLUSTERED ([Id] ASC),
    CONSTRAINT [FK1] FOREIGN KEY ([Bill_no]) REFERENCES [dbo].[BillTbl] ([BillId])
);




Table 4: ProductTbl

CREATE TABLE [dbo].[ProductTbl] (
    [PrId]      INT           IDENTITY (100, 1) NOT NULL,
    [PrName]    VARCHAR (MAX) NOT NULL,
    [PrCat]     INT           NOT NULL,
    [PrPrice]   INT           NOT NULL,
    [PrQty]     INT           NOT NULL,
    [PrExpDate] DATE          NOT NULL,
    [PrDesc]    VARCHAR (MAX) NOT NULL,
    [PrImg]     VARCHAR (MAX) NOT NULL,
    CONSTRAINT [PK_ProductTbl] PRIMARY KEY CLUSTERED ([PrId] ASC),
    CONSTRAINT [FK_Category] FOREIGN KEY ([PrCat]) REFERENCES [dbo].[CategoryTbl] ([CatId])
);




Table 5: SellerTbl

CREATE TABLE [dbo].[SellerTbl] (
    [SelId]       INT          IDENTITY (100, 1) NOT NULL,
    [SelName]     VARCHAR (50) NOT NULL,
    [SelEmail]    VARCHAR (50) NOT NULL,
    [SelPassword] VARCHAR (50) NOT NULL,
    [SelPhone]    VARCHAR (50) NOT NULL,
    [SelAddress]  VARCHAR (50) NOT NULL,
    CONSTRAINT [PK_SellerTbl] PRIMARY KEY CLUSTERED ([SelId] ASC)
);
